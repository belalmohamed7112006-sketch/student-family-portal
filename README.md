## خطوة قاعدة البيانات المطلوبة
من Supabase افتح **SQL Editor** ثم نفّذ النص التالي مرة واحدة. هذا ينشئ/يكمل الجداول وسياسات الصلاحيات، ويضيف Trigger ينشئ ملف العضو تلقائيًا عند التسجيل حتى لو كان تأكيد البريد مفعّلًا.

```sql
create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  username text,
  full_name text,
  birth_date date,
  national_id text,
  college text,
  academic_year text,
  role text not null default 'member' check (role in ('member','admin')),
  approval_status text not null default 'pending' check (approval_status in ('pending','approved','rejected')),
  created_at timestamptz not null default now()
);

alter table public.profiles add column if not exists username text;
alter table public.profiles add column if not exists full_name text;
alter table public.profiles add column if not exists birth_date date;
alter table public.profiles add column if not exists national_id text;
alter table public.profiles add column if not exists college text;
alter table public.profiles add column if not exists academic_year text;
alter table public.profiles add column if not exists role text not null default 'member';
alter table public.profiles add column if not exists approval_status text not null default 'pending';
alter table public.profiles add column if not exists created_at timestamptz not null default now();

create table if not exists public.activity_submissions (
  id bigint generated by default as identity primary key,
  user_id uuid not null references public.profiles(id) on delete cascade,
  category text not null check (category in ('event','military_visit','field_visit','workshop')),
  activity_name text not null,
  activity_date date,
  approval_status text not null default 'pending' check (approval_status in ('pending','approved','rejected')),
  created_at timestamptz not null default now()
);

alter table public.activity_submissions add column if not exists approval_status text not null default 'pending';
alter table public.activity_submissions add column if not exists created_at timestamptz not null default now();

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
begin
  insert into public.profiles
    (id, username, full_name, birth_date, national_id, college, academic_year, role, approval_status)
  values
    (
      new.id,
      new.raw_user_meta_data ->> 'username',
      new.raw_user_meta_data ->> 'full_name',
      nullif(new.raw_user_meta_data ->> 'birth_date','')::date,
      new.raw_user_meta_data ->> 'national_id',
      new.raw_user_meta_data ->> 'college',
      new.raw_user_meta_data ->> 'academic_year',
      'member',
      'pending'
    )
  on conflict (id) do update set
    username = excluded.username,
    full_name = excluded.full_name,
    birth_date = excluded.birth_date,
    national_id = excluded.national_id,
    college = excluded.college,
    academic_year = excluded.academic_year;
  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
after insert on auth.users
for each row execute procedure public.handle_new_user();

alter table public.profiles enable row level security;
alter table public.activity_submissions enable row level security;

drop policy if exists "profiles_select_self_or_admin" on public.profiles;
create policy "profiles_select_self_or_admin" on public.profiles
for select to authenticated
using (
  id = (select auth.uid())
  or exists (
    select 1 from public.profiles me
    where me.id = (select auth.uid())
      and me.role = 'admin'
      and me.approval_status = 'approved'
  )
);

drop policy if exists "profiles_update_admin" on public.profiles;
create policy "profiles_update_admin" on public.profiles
for update to authenticated
using (
  exists (
    select 1 from public.profiles me
    where me.id = (select auth.uid())
      and me.role = 'admin'
      and me.approval_status = 'approved'
  )
)
with check (
  exists (
    select 1 from public.profiles me
    where me.id = (select auth.uid())
      and me.role = 'admin'
      and me.approval_status = 'approved'
  )
);

drop policy if exists "activities_select_owner_or_admin" on public.activity_submissions;
create policy "activities_select_owner_or_admin" on public.activity_submissions
for select to authenticated
using (
  user_id = (select auth.uid())
  or exists (
    select 1 from public.profiles me
    where me.id = (select auth.uid())
      and me.role = 'admin'
      and me.approval_status = 'approved'
  )
);

drop policy if exists "activities_insert_approved_member" on public.activity_submissions;
create policy "activities_insert_approved_member" on public.activity_submissions
for insert to authenticated
with check (
  user_id = (select auth.uid())
  and exists (
    select 1 from public.profiles me
    where me.id = (select auth.uid())
      and me.approval_status = 'approved'
      and me.role = 'member'
  )
);

drop policy if exists "activities_update_admin" on public.activity_submissions;
create policy "activities_update_admin" on public.activity_submissions
for update to authenticated
using (
  exists (
    select 1 from public.profiles me
    where me.id = (select auth.uid())
      and me.role = 'admin'
      and me.approval_status = 'approved'
  )
)
with check (
  exists (
    select 1 from public.profiles me
    where me.id = (select auth.uid())
      and me.role = 'admin'
      and me.approval_status = 'approved'
  )
);
```

## تعيين حساب المدير الحالي
بعد تنفيذ SQL، افتح **Table Editor → profiles** وابحث عن حساب الإدارة الحالي، ثم اجعل:
- `role` = `admin`
- `approval_status` = `approved`

لا تجعل أي حساب آخر مديرًا.

## إعداد تأكيد البريد
لو خيار تأكيد البريد الإلكتروني مفعّل في Supabase، المستخدم سيحتاج فتح رسالة التأكيد. ملف العضوية ينشأ تلقائيًا عن طريق Trigger عند إنشاء مستخدم Auth. اضبط رابط الموقع المنشور في **Authentication → URL Configuration → Site URL**، وأضف رابط Vercel إلى **Redirect URLs**.

## تنبيه أمان
المفتاح الموجود في `config.js` هو Publishable key مخصص للمتصفح. لا تضع أبدًا `service_role` أو أي Secret key داخل ملفات الموقع.
